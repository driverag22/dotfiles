*** org-timegrid (agenda calendar)
#+BEGIN_SRC emacs-lisp
    (setq calendar-week-start-day 1)
    (setq org-agenda-start-on-weekday 1)

    (unless (package-installed-p 'org-timegrid)
      (package-vc-install "https://github.com/Gleek/org-timegrid.git"))

    (use-package org-timegrid
      :commands (org-timegrid-week)
      :demand t
      :bind ("C-c a" . org-timegrid-week)
      :init
      ;; The symbol `agenda' means: read events from `org-agenda-files'.
      (setq org-timegrid-org-files 'agenda
            org-timegrid-default-zoom 0.7
  		  org-timegrid-org-capture-file (expand-file-name "calendar.org" "~/org/")
            org-timegrid-org-capture-template '(:target file :template "* %{title}\n%{time-range}\n%?")
            ;; Save Org buffers immediately after edits made in the calendar.
            org-timegrid-org-auto-save t
            ;; Set this to nil if repeating entries should be hidden.
            org-timegrid-org-show-repeaters t
        	)
      :config
      (require 'org-timegrid-org)
      
      ;; --- EVIL KEYBINDINGS ---
      (with-eval-after-load 'evil
        (evil-set-initial-state 'org-timegrid-mode 'normal)
        (evil-define-key 'normal org-timegrid-mode-map
          "l" 'org-timegrid-cursor-forward-day
          "h" 'org-timegrid-cursor-backward-day
      ))
  )
